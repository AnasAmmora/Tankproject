// Unity MCP Editor method body. Generates a single closed low-poly lathed mesh.
if(UnityEditor.EditorApplication.isPlaying)throw new System.Exception("Requires Edit Mode.");
string folder="Assets/ProjectsFiles/Art/Projectiles";
foreach(string path in new string[]{"Assets/ProjectsFiles/Art",folder})
    if(!UnityEditor.AssetDatabase.IsValidFolder(path))UnityEditor.AssetDatabase.CreateFolder(System.IO.Path.GetDirectoryName(path).Replace('\\','/'),System.IO.Path.GetFileName(path));
if(UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Mesh>(folder+"/CartoonShell.asset")!=null)
    throw new System.Exception("Shell asset already exists; inspect before regenerating.");
var shader=UnityEngine.Shader.Find("Universal Render Pipeline/Lit");
if(shader==null)throw new System.Exception("URP Lit unavailable.");
var olive=new UnityEngine.Material(shader){name="Shell Olive Matte"};
olive.SetColor("_BaseColor",new UnityEngine.Color(0.31f,0.37f,0.17f,1));olive.SetFloat("_Metallic",0f);olive.SetFloat("_Smoothness",0.16f);
var metal=new UnityEngine.Material(shader){name="Shell Gray Metal Matte"};
metal.SetColor("_BaseColor",new UnityEngine.Color(0.46f,0.50f,0.52f,1));metal.SetFloat("_Metallic",0.25f);metal.SetFloat("_Smoothness",0.23f);
UnityEditor.AssetDatabase.CreateAsset(olive,folder+"/ShellOlive.mat");
UnityEditor.AssetDatabase.CreateAsset(metal,folder+"/ShellMetal.mat");
// z/r silhouette, base (-Z) to rounded tip (+Z). Band and rim are part of
// the same continuous surface; no intersecting cylinders or internal faces.
float[] z={-0.50f,-0.48f,-0.435f,-0.415f,-0.385f,-0.35f,-0.32f,0.16f,0.21f,0.34f,0.435f,0.48f,0.495f,0.50f};
float[] radius={0.185f,0.205f,0.205f,0.173f,0.173f,0.18f,0.173f,0.173f,0.166f,0.121f,0.065f,0.031f,0.014f,0.006f};
const int sides=12;
var vertices=new System.Collections.Generic.List<UnityEngine.Vector3>();
var uv=new System.Collections.Generic.List<UnityEngine.Vector2>();
var body=new System.Collections.Generic.List<int>();var metallic=new System.Collections.Generic.List<int>();
System.Func<int,int,UnityEngine.Vector3> point=(row,side)=>{float a=2*UnityEngine.Mathf.PI*side/sides;return new UnityEngine.Vector3(radius[row]*UnityEngine.Mathf.Cos(a),radius[row]*UnityEngine.Mathf.Sin(a),z[row]);};
for(int row=0;row<z.Length-1;row++)for(int side=0;side<sides;side++)
{
    int first=vertices.Count;
    vertices.Add(point(row,side));vertices.Add(point(row,side+1));vertices.Add(point(row+1,side+1));vertices.Add(point(row+1,side));
    uv.Add(new UnityEngine.Vector2((float)side/sides,z[row]+0.5f));uv.Add(new UnityEngine.Vector2((float)(side+1)/sides,z[row]+0.5f));uv.Add(new UnityEngine.Vector2((float)(side+1)/sides,z[row+1]+0.5f));uv.Add(new UnityEngine.Vector2((float)side/sides,z[row+1]+0.5f));
    var triangles=row<4 || row>=7?metallic:body;
    triangles.AddRange(new int[]{first,first+1,first+2,first,first+2,first+3});
}
foreach(int row in new int[]{0,z.Length-1})for(int side=0;side<sides;side++)
{
    int first=vertices.Count;vertices.Add(new UnityEngine.Vector3(0,0,z[row]));vertices.Add(point(row,side));vertices.Add(point(row,side+1));
    uv.Add(new UnityEngine.Vector2(0.5f,0.5f));uv.Add(UnityEngine.Vector2.zero);uv.Add(UnityEngine.Vector2.one);
    metallic.Add(first);metallic.Add(first+(row==0?2:1));metallic.Add(first+(row==0?1:2));
}
var mesh=new UnityEngine.Mesh(){name="Cartoon Shell 12 Sided"};mesh.SetVertices(vertices);mesh.SetUVs(0,uv);mesh.subMeshCount=2;mesh.SetTriangles(body,0);mesh.SetTriangles(metallic,1);mesh.RecalculateNormals();mesh.RecalculateBounds();
UnityEditor.AssetDatabase.CreateAsset(mesh,folder+"/CartoonShell.asset");
var visual=new UnityEngine.GameObject("Shell Visual");
try
{
    visual.AddComponent<UnityEngine.MeshFilter>().sharedMesh=mesh;
    var renderer=visual.AddComponent<UnityEngine.MeshRenderer>();renderer.sharedMaterials=new UnityEngine.Material[]{olive,metal};
    renderer.shadowCastingMode=UnityEngine.Rendering.ShadowCastingMode.Off;renderer.receiveShadows=true;
    UnityEditor.PrefabUtility.SaveAsPrefabAsset(visual,folder+"/ShellVisual.prefab");
}
finally{UnityEngine.Object.DestroyImmediate(visual);}
string bulletPath="Assets/ProjectsFiles/Prefabes/Bullet.prefab";
System.IO.File.WriteAllText("Tests/ShellVisual/bullet-before.prefab.txt",System.IO.File.ReadAllText(bulletPath));
var root=UnityEditor.PrefabUtility.LoadPrefabContents(bulletPath);
try
{
    var rb=root.GetComponent<UnityEngine.Rigidbody>();var collider=root.GetComponent<UnityEngine.Collider>();var script=root.GetComponent<Bullet>();
    string rbBefore=UnityEditor.EditorJsonUtility.ToJson(rb), colliderBefore=UnityEditor.EditorJsonUtility.ToJson(collider),scriptBefore=UnityEditor.EditorJsonUtility.ToJson(script);
    var scaleBefore=root.transform.localScale;
    UnityEngine.Object.DestroyImmediate(root.GetComponent<UnityEngine.MeshRenderer>());
    UnityEngine.Object.DestroyImmediate(root.GetComponent<UnityEngine.MeshFilter>());
    var child=(UnityEngine.GameObject)UnityEditor.PrefabUtility.InstantiatePrefab(UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.GameObject>(folder+"/ShellVisual.prefab"),root.transform);
    child.transform.localPosition=UnityEngine.Vector3.zero;child.transform.localRotation=UnityEngine.Quaternion.identity;child.transform.localScale=UnityEngine.Vector3.one;child.layer=root.layer;
    UnityEditor.PrefabUtility.SaveAsPrefabAsset(root,bulletPath);
    if(rbBefore!=UnityEditor.EditorJsonUtility.ToJson(rb)||colliderBefore!=UnityEditor.EditorJsonUtility.ToJson(collider)||scriptBefore!=UnityEditor.EditorJsonUtility.ToJson(script)||root.transform.localScale!=scaleBefore)
        throw new System.Exception("Gameplay data changed unexpectedly.");
    var report="Saved continuous shell mesh: "+mesh.vertexCount+" vertices / "+(body.Count+metallic.Count)/3+" triangles / 2 submeshes and matte materials. Local bounds="+mesh.bounds+"; gameplay bounds size="+UnityEngine.Vector3.Scale(mesh.bounds.size,scaleBefore)+"; +Z tip. Rigidbody/Collider/Bullet JSON and root scale unchanged. Visual has only Transform/MeshFilter/MeshRenderer.";
    System.IO.File.WriteAllText("Tests/ShellVisual/setup.txt",report);return report;
}
finally{UnityEditor.PrefabUtility.UnloadPrefabContents(root);UnityEditor.AssetDatabase.SaveAssets();}
